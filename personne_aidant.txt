# backend/models/aidant_relation.py
from pydantic import BaseModel, EmailStr
from typing import Optional, List
from datetime import datetime

class AidantRelation(BaseModel):
    id: Optional[str]
    aide_id: str  # l'utilisateur aidé
    aidant_id: str  # l'utilisateur qui aide
    permissions: dict  # ex: {"can_edit": True, "can_alert": True}
    status: str = "pending"  # pending | accepted | revoked
    created_at: datetime = datetime.utcnow()

🛡️ Sécurité et gestion d’accès
⚠️ Tous les endpoints /aidant/... vérifient bien que current_user.id == aidant_id d'une relation acceptée.
🔐 Les tâches marquées private: true ne sont pas accessibles à l’aidant.
🧠 Logs d’accès aux tâches pour suivi.
